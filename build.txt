#build:

nasm -f bin boot.asm -o boot.bin

g++ -m32 -ffreestanding -c kernel.cpp -o kernel.o

ld -mi386pe -T linker.ld -o kernel.exe kernel.o

objcopy -O binary kernel.exe kernel.bin         

cmd /c copy /b boot.bin+kernel.bin koyot.img

#run in qemu :
 qemu-system-x86_64 -drive format=raw,file=koyot.img
