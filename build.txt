#build:

nasm -f bin boot.asm -o boot.bin

g++ -m32 -ffreestanding -c kernel.cpp -o kernel.o

ld -m elf_i386 -T linker.ld -o kernel.elf kernel.o

objcopy -O binary kernel.elf kernel.bin         

python -c "open('koyot.img','wb').write(open('boot.bin','rb').read()+open('kernel.bin','rb').read())"

#run in qemu :
 qemu-system-x86_64 -drive format=raw,file=koyot.img
